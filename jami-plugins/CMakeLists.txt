cmake_minimum_required(VERSION 3.10)
project(jami-plugins)

if(NOT DEFINED PLATFORM_TYPE)
    set(PLATFORM_TYPE "LINUX")
endif()

# Probe whether daemon contrib pkgconfig is present — needed by ffmpeg-dependent plugins.
# Each plugin overrides PKG_CONFIG_LIBDIR to the contrib path, so system ffmpeg is bypassed.
# Derive the expected path from PLATFORM_TYPE rather than hardcoding the triplet.
if(PLATFORM_TYPE STREQUAL "LINUX")
    set(_contrib_pkgconfig "${CMAKE_CURRENT_SOURCE_DIR}/daemon/contrib/x86_64-linux-gnu/lib/pkgconfig")
elseif(PLATFORM_TYPE STREQUAL "MACOS")
    execute_process(COMMAND sh -c "cc -dumpmachine" OUTPUT_VARIABLE _host_triplet OUTPUT_STRIP_TRAILING_WHITESPACE)
    set(_contrib_pkgconfig "${CMAKE_CURRENT_SOURCE_DIR}/daemon/contrib/${_host_triplet}/lib/pkgconfig")
    unset(_host_triplet)
else()
    set(_contrib_pkgconfig "")
endif()
if(_contrib_pkgconfig AND EXISTS "${_contrib_pkgconfig}")
    set(FFMPEG_IN_CONTRIB TRUE)
else()
    set(FFMPEG_IN_CONTRIB FALSE)
    message(STATUS "Daemon contrib pkgconfig not found — AudioFilter, TestSuite (and ffmpeg-using plugins) skipped. Build daemon/contrib to enable.")
endif()
unset(_contrib_pkgconfig)

if(FFMPEG_IN_CONTRIB)
    add_subdirectory(AudioFilter)
endif()
add_subdirectory(AutoAnswer)
# OpenCV-dependent plugins: probe quietly so a missing OpenCV doesn't abort configure
find_package(OpenCV 4 QUIET)
if(OpenCV_FOUND AND FFMPEG_IN_CONTRIB)
    # Example is a template plugin (FILL.cpp placeholder) — not built
    add_subdirectory(GreenScreen)
    add_subdirectory(HelloWorld)
    add_subdirectory(Segmentation)
elseif(NOT OpenCV_FOUND)
    message(STATUS "OpenCV 4 not found — GreenScreen, HelloWorld, Segmentation skipped")
elseif(NOT FFMPEG_IN_CONTRIB)
    message(STATUS "Daemon contrib missing — GreenScreen, HelloWorld, Segmentation skipped")
endif()
if(WIN32)
    add_subdirectory(TensorflowSegmentation)
endif()
if(FFMPEG_IN_CONTRIB)
    add_subdirectory(TestSuite)
endif()
option(BUILD_TRANSLATE_MESSAGES "Build TranslateMessages plugin (requires ORT, SentencePiece, fastText, ICU)" OFF)
if(BUILD_TRANSLATE_MESSAGES)
    # When building as a subdirectory of the jami workspace the daemon lives one
    # level above jami-plugins (not inside it).  Detect this and propagate DAEMON
    # so TranslateMessages does not fall back to jami-plugins/daemon/contrib which
    # may not have the full set of built contrib headers.
    if(NOT DEFINED DAEMON AND NOT CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
        get_filename_component(_ws_daemon "${CMAKE_CURRENT_SOURCE_DIR}/../daemon" ABSOLUTE)
        if(EXISTS "${_ws_daemon}/contrib")
            set(DAEMON "${_ws_daemon}")
        endif()
        unset(_ws_daemon)
    endif()
    # Guard: setup_contrib.cmake aborts if contrib headers are absent. Skip gracefully.
    if(DEFINED DAEMON)
        set(_tm_contrib_include "${DAEMON}/contrib/x86_64-linux-gnu/include")
    else()
        set(_tm_contrib_include "${CMAKE_CURRENT_SOURCE_DIR}/daemon/contrib/x86_64-linux-gnu/include")
    endif()
    if(EXISTS "${_tm_contrib_include}")
        add_subdirectory(TranslateMessages)
    else()
        message(STATUS "BUILD_TRANSLATE_MESSAGES=ON but daemon contrib headers not found at ${_tm_contrib_include} — TranslateMessages skipped. Build daemon/contrib first.")
    endif()
    unset(_tm_contrib_include)
endif()
add_subdirectory(WaterMark)
option(BUILD_WHISPER_TRANSCRIPT "Build WhisperTranscript plugin (requires pre-built libwhisper in contrib)" OFF)
if(BUILD_WHISPER_TRANSCRIPT)
    add_subdirectory(WhisperTranscript)
endif()
